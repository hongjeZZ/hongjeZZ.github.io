```bash
# 설치
brew install fzf                       # macOS
sudo apt install fzf                   # Debian 9+ / Ubuntu 19.10+
git clone --depth 1 https://github.com/junegunn/fzf.git ~/.fzf && ~/.fzf/install   # 패키지 관리자 없이
fzf --version
```

```bash
# 기본 사용 — 표준 입력의 목록에서 고른 줄을 표준 출력으로 낸다
fzf                                    # 입력이 없으면 현재 디렉터리 아래 파일을 훑는다
find * -type f | fzf > selected        # 고른 줄을 파일로
vim $(fzf)                             # 고른 파일을 바로 연다
fzf --print0 | xargs -0 -o vim         # 공백이 든 파일 이름도 안전하게 넘긴다
fzf --bind 'enter:become(vim {})'      # Enter를 누르면 fzf가 vim으로 바뀐다
```

```bash
# 검색 문법 — 입력창에 치는 검색어, 공백으로 나눈 검색어는 모두 맞아야 한다
fzf -q 'sbtrkt'                        # 퍼지: 글자가 순서대로 들어 있으면 맞음
fzf -q "'wild"                         # wild를 그대로 포함
fzf -q "'wild'"                        # 단어 경계에서 wild와 일치
fzf -q '^music'                        # music으로 시작
fzf -q '.mp3$'                         # .mp3로 끝남
fzf -q '!fire'                         # fire를 포함하지 않음
fzf -q '!^music'                       # music으로 시작하지 않음
fzf -q '!.mp3$'                        # .mp3로 끝나지 않음
fzf -q '^core go$ | rb$ | py$'         # |는 OR: core로 시작하고 go·rb·py 중 하나로 끝남
```

```bash
# 자주 쓰는 옵션
fzf -m                                 # 여러 개 고르기 (TAB / Shift-TAB)
fzf -e                                 # 퍼지 대신 정확 매치 (이때 '를 붙인 검색어는 퍼지)
fzf -i                                 # 대소문자 무시 (기본은 검색어에 대문자가 있을 때만 구분)
fzf -q init -1 -0                      # 검색어를 넣고 시작, 하나만 맞으면 바로 고르고 없으면 바로 끝냄
fzf -f init                            # 화면 없이 걸러 낸 결과만 출력
fzf --height 40% --layout reverse --border   # 전체 화면 대신 커서 아래 40%에 띄움
fzf --preview 'cat {}'                 # 현재 줄({})을 오른쪽에 미리보기
fzf --preview 'cat {}' --preview-window down,50%   # 미리보기를 아래쪽 절반에
fzf --tac +s                           # 입력 순서를 뒤집고 정렬하지 않음
ps -ef | fzf --header-lines 1          # 첫 줄은 헤더로 고정
ps -ef | fzf --nth 8..                 # 8번째 필드부터만 검색 (필드는 공백으로 나눔)
fzf -d : --with-nth 1 --accept-nth 1,7 < /etc/passwd   # :로 나눠 1번 필드만 보여 주고, 고르면 1·7번 출력
```

```bash
# 찾기 화면 키 — fzf가 떠 있을 때
CTRL-K / CTRL-J                        # 위 / 아래 (CTRL-P / CTRL-N, ↑ / ↓도 된다)
Enter                                  # 고르기
TAB / Shift-TAB                        # -m일 때 표시하고 아래 / 위로
CTRL-C / CTRL-G / ESC                  # 나가기
CTRL-U / CTRL-W                        # 검색어 지우기 / 한 단어 지우기
CTRL-/                                 # 긴 줄 줄바꿈 켜고 끄기
Shift-↑ / Shift-↓                      # 미리보기 스크롤
fzf --bind 'ctrl-y:execute-silent(echo -n {} | pbcopy)+abort'   # 키에 동작 붙이기: CTRL-Y로 복사하고 닫기
```

```bash
# 셸 통합 켜기 — 셸 설정 파일에 한 줄
eval "$(fzf --bash)"                   # ~/.bashrc
source <(fzf --zsh)                    # ~/.zshrc
fzf --fish | source                    # ~/.config/fish/config.fish
FZF_CTRL_R_COMMAND= FZF_ALT_C_COMMAND= source <(fzf --zsh)   # CTRL-R·ALT-C는 빼고 켜기 (불러온 뒤에 비우면 소용없다)
```

```bash
# 셸 통합 키
CTRL-T                                 # 고른 파일·디렉터리를 명령줄에 붙여 넣기 (TAB으로 여러 개)
CTRL-R                                 # 명령 기록에서 골라 붙여 넣기 (한 번 더 누르면 시간순 ↔ 관련도순)
ALT-C                                  # 고른 디렉터리로 cd
```

```bash
# ** 자동완성 (bash·zsh) — 명령 뒤에 **를 붙이고 TAB
vim **<TAB>                            # 현재 디렉터리 아래 파일 (TAB으로 여러 개)
vim ../fzf**<TAB>                      # 상위 디렉터리 아래에서 fzf가 들어간 파일
cd **<TAB>                             # 디렉터리
kill -9 **<TAB>                        # 프로세스 PID
ssh **<TAB>                            # /etc/hosts, ~/.ssh/config의 호스트 이름
export **<TAB>                         # 환경 변수 (unset, unalias도 된다)
```

```bash
# 기본값 — 셸 설정 파일에
export FZF_DEFAULT_COMMAND='fd --type f'   # 입력 없이 띄울 때 목록을 만들 명령, 여기선 fd (셸 통합 키에는 안 쓰인다)
export FZF_DEFAULT_OPTS='--height 40% --layout reverse --border'   # 늘 붙일 옵션
export FZF_CTRL_T_OPTS="--preview 'cat {}'"   # CTRL-T에만 붙일 옵션 (FZF_CTRL_R_OPTS, FZF_ALT_C_OPTS도 있다)
export FZF_COMPLETION_TRIGGER='~~'     # ** 대신 쓸 자동완성 트리거
```

```bash
# 다른 명령과 조합
git switch $(git branch --format='%(refname:short)' | fzf)          # 브랜치 골라 전환
git log --oneline | fzf --preview 'git show --color=always {1}'     # 커밋 고르며 내용 미리보기 ({1}은 첫 필드)
ps -ef | fzf -m --header-lines 1 --accept-nth 2 | xargs kill        # 프로세스 골라 종료
cd "$(find . -type d | fzf)"                                        # 디렉터리 골라 이동
```
