```bash
# 설치·세팅 — 아래 export와 source는 ~/.zshrc에
brew install fzf ripgrep                                # Debian·Ubuntu는 sudo apt install fzf ripgrep
export FZF_DEFAULT_COMMAND='rg --files --hidden --glob "!.git"'   # 파일 목록을 rg로 (.gitignore 따름)
export FZF_CTRL_T_COMMAND="$FZF_DEFAULT_COMMAND"        # CTRL-T도 같은 목록 (기본 명령은 셸 통합 키에 안 쓰인다)
export FZF_DEFAULT_OPTS='--height 40% --layout reverse' # 전체 화면 대신 커서 아래 40%에
source <(fzf --zsh)                                     # 셸 통합 켜기 (bash는 eval "$(fzf --bash)")
```

```bash
# 셸 통합 키
CTRL-T                                                  # 파일 골라 명령줄에 붙여 넣기 (TAB으로 여러 개)
CTRL-R                                                  # 명령 기록에서 골라 붙여 넣기
ALT-C                                                   # 디렉터리 골라 cd
vim **<TAB>                                             # 파일 자동완성 (cd는 디렉터리, kill은 PID, ssh는 호스트)
```

```bash
# 검색 문법 — 입력창에 친다 (-q는 미리 넣고 시작), 공백으로 나누면 모두 맞아야 한다
fzf -q "'wild"                                          # wild를 그대로 포함
fzf -q '^src'                                           # src로 시작
fzf -q '.go$'                                           # .go로 끝남
fzf -q '!test'                                          # test를 포함하지 않음
fzf -q '.go$ | .py$'                                    # .go 또는 .py로 끝남
```

```bash
# 자주 쓰는 조합
vim $(fzf -m)                                           # 파일 골라 에디터로 열기 (TAB으로 여러 개)
fzf --preview 'cat {}'                                  # 고르면서 내용 미리보기 ({}는 현재 줄)
git switch $(git branch --format='%(refname:short)' | fzf)   # 브랜치 골라 전환
```
